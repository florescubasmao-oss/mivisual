function mv528ValorComparable_(valor) {
  if (valor instanceof Date) return "D:" + valor.getTime();
  if (valor === null || valor === undefined) return "";
  if (typeof valor === "number") return "N:" + String(valor);
  if (typeof valor === "boolean") return "B:" + (valor ? "1" : "0");
  return "S:" + String(valor);
}

function mv528FilaIgual_(a,b,columnas) {
  for (let c=0;c<columnas;c++) {
    if (mv528ValorComparable_(a && a[c]) !== mv528ValorComparable_(b && b[c])) return false;
  }
  return true;
}

function mv528TramosCambiados_(anteriores,nuevas,columnas,maxHueco) {
  const limite=Math.min(anteriores.length,nuevas.length);
  const indices=[];
  for (let i=0;i<limite;i++) {
    if (!mv528FilaIgual_(anteriores[i],nuevas[i],columnas)) indices.push(i);
  }
  for (let i=limite;i<nuevas.length;i++) indices.push(i);
  if (!indices.length) return [];

  const hueco=Math.max(0,Number(maxHueco)||0);
  const tramos=[];
  let inicio=indices[0], fin=indices[0];
  for (let k=1;k<indices.length;k++) {
    const actual=indices[k];
    if (actual-fin-1<=hueco) fin=actual;
    else { tramos.push([inicio,fin]); inicio=fin=actual; }
  }
  tramos.push([inicio,fin]);
  return tramos;
}

function mv528PersistirDiferencias_(hoja,anteriores,nuevas,columnas,opciones) {
  opciones=opciones||{};
  const tramos=mv528TramosCambiados_(anteriores,nuevas,columnas,opciones.maxHueco==null?3:opciones.maxHueco);
  let filasEscritas=0;

  tramos.forEach(function(tramo){
    const inicio=tramo[0], fin=tramo[1];
    const cantidad=fin-inicio+1;
    hoja.getRange(inicio+2,1,cantidad,columnas).setValues(nuevas.slice(inicio,fin+1));
    filasEscritas+=cantidad;
    if (typeof opciones.formatearTramo === "function") {
      opciones.formatearTramo(hoja,inicio+2,cantidad);
    }
  });

  if (anteriores.length>nuevas.length) {
    hoja.getRange(nuevas.length+2,1,anteriores.length-nuevas.length,columnas).clearContent();
  }
  return {tramos:tramos.length,filasEscritas:filasEscritas,filasEliminadas:Math.max(0,anteriores.length-nuevas.length)};
}
