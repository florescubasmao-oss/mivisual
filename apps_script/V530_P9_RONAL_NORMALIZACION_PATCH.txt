function normalizarCuadrilla(nombre) {
  return (nombre || "")
    .toString()
    .replace(/^P\s+(\d+)/i, "P$1")
    .replace(/P9\s+VISUAL\s+SGA\s+RONALD\s+ENRIQUE\s+TESEN\s+CALDERON/gi,
             "P9 VISUAL SGA RONAL ENRIQUE TESEN CALDERON")
    .replace(/\s+/g, " ")
    .trim();
}
